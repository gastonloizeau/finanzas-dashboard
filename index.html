const fm = FileManager.iCloud() const path = fm.joinPath(fm.documentsDirectory(), "Finanzas/dashboard.json") await fm.downloadFileFromiCloud(path) const raw = fm.readString(path) const d = JSON.parse(raw) const fmt = (n) => { const abs = Math.abs(n) const sign = n < 0 ? '-' : '' if (abs >= 1000000) return sign + '$' + (abs/1000000).toFixed(1) + 'M' if (abs >= 1000) return sign + '$' + Math.round(abs/1000) + 'k' return sign + '$' + abs } const w = new ListWidget() w.backgroundColor = new Color("#1a1a1a") w.setPadding(12, 14, 12, 14) // Header const header = w.addStack() header.layoutHorizontally() const title = header.addText("Mayo 2026") title.font = Font.mediumSystemFont(11) title.textColor = new Color("#888888") header.addSpacer() const dolar = header.addText("MEP $" + d.dolar_mep.toLocaleString('es-AR')) dolar.font = Font.systemFont(10) dolar.textColor = new Color("#888888") w.addSpacer(6) // Total del mes const totalText = w.addText(fmt(d.total_mes)) totalText.font = Font.boldSystemFont(26) totalText.textColor = d.total_mes < 0 ? new Color("#E24B4A") : new Color("#1D9E75") const totalLabel = w.addText("Total del mes") totalLabel.font = Font.systemFont(10) totalLabel.textColor = new Color("#666666") w.addSpacer(8) // Fila ingresos / gastos const row1 = w.addStack() row1.layoutHorizontally() const ingStack = row1.addStack() ingStack.layoutVertically() const ingVal = ingStack.addText(fmt(d.ingresos)) ingVal.font = Font.mediumSystemFont(13) ingVal.textColor = new Color("#1D9E75") const ingLabel = ingStack.addText("Ingresos") ingLabel.font = Font.systemFont(9) ingLabel.textColor = new Color("#666666") row1.addSpacer() const gasStack = row1.addStack() gasStack.layoutVertically() const gasVal = gasStack.addText(fmt(d.gastos)) gasVal.font = Font.mediumSystemFont(13) gasVal.textColor = new Color("#E24B4A") const gasLabel = gasStack.addText("Gastos") gasLabel.font = Font.systemFont(9) gasLabel.textColor = new Color("#666666") row1.addSpacer() const musStack = row1.addStack() musStack.layoutVertically() const musVal = musStack.addText(fmt(d.musica_neta)) musVal.font = Font.mediumSystemFont(13) musVal.textColor = new Color("#7F77DD") const musLabel = musStack.addText("Música") musLabel.font = Font.systemFont(9) musLabel.textColor = new Color("#666666") w.addSpacer(8) // Patrimonio const patRow = w.addStack() patRow.layoutHorizontally() const patVal = patRow.addText("USD " + d.patrimonio_usd.toLocaleString('es-AR')) patVal.font = Font.mediumSystemFont(12) patVal.textColor = new Color("#ffffff") patRow.addSpacer() const patLabel = patRow.addText("Patrimonio") patLabel.font = Font.systemFont(10) patLabel.textColor = new Color("#666666") w.addSpacer(4) // Fecha const fecha = w.addText("Actualizado " + d.fecha) fecha.font = Font.systemFont(9) fecha.textColor = new Color("#444444") Script.setWidget(w) Script.complete() w.presentMedium()
